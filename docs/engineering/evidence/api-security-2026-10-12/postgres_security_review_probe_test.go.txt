package postgres

import (
 "context"
 "testing"
 "time"
 "github.com/joseantoniogarciay/TournamentsManager/apps/backend/internal/tournaments"
)

// Audit probe: characterize the current behavior, not the desired invariant.
func TestIntegrationSecurityReviewInvitationRevocationWhileJoinWaits(t *testing.T) {
 ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
 defer cancel()
 pool := integrationPool(t)
 owner := createVerifiedLocalAccount(ctx,t,pool,"review-owner@example.test","reviewowner","correct password")
 participant := createVerifiedLocalAccount(ctx,t,pool,"review-player@example.test","reviewplayer","correct password")
 service := tournaments.NewCreationService(NewAccountTournamentRepository(pool))
 tournament,err := service.Create(ctx,owner,tournaments.CreateInput{Name:"Audit",Sport:tournaments.SportFootball,Teams:[]tournaments.TeamInput{{Name:"Owner"}}})
 if err != nil { t.Fatal(err) }
 token,err := service.CreateTeamInvitation(ctx,owner,tournament.ID)
 if err != nil { t.Fatal(err) }
 tx,err := pool.Begin(ctx)
 if err != nil { t.Fatal(err) }
 defer func(){ _ = tx.Rollback(context.Background()) }()
 if _,err = tx.Exec(ctx,`SELECT id FROM tournaments WHERE id=$1 FOR UPDATE`,tournament.ID); err != nil { t.Fatal(err) }
 outcome := make(chan error,1)
 go func(){ _,err := service.JoinTeamInvitation(ctx,participant,token,tournaments.TeamInput{Name:"Player"}); outcome <- err }()
 for {
  var blocked bool
  if err := pool.QueryRow(ctx,`SELECT EXISTS(SELECT 1 FROM pg_stat_activity WHERE datname=current_database() AND wait_event_type='Lock' AND query LIKE '%FOR UPDATE OF tournaments%')`).Scan(&blocked); err != nil { t.Fatal(err) }
  if blocked { break }
  select { case err := <-outcome: t.Fatalf("join did not wait: %v",err); case <-ctx.Done(): t.Fatal(ctx.Err()); case <-time.After(10*time.Millisecond): }
 }
 if _,err = tx.Exec(ctx,`DELETE FROM tournament_team_invitations WHERE tournament_id=$1`,tournament.ID); err != nil { t.Fatal(err) }
 if err = tx.Commit(ctx); err != nil { t.Fatal(err) }
 select {
 case err := <-outcome:
  if err != nil { t.Fatalf("probe behavior changed; inspect: %v",err) }
  t.Log("CONFIRMED: join succeeded after invitation revocation committed while waiting for tournament lock")
 case <-ctx.Done(): t.Fatal(ctx.Err())
 }
}
